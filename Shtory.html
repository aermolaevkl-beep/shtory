            if (heightGroup === 'свыше2.0') {

                const extraCm = height - 200;

                const steps = Math.ceil(extraCm / 10);

                extra = basePrice * 0.1 * steps;

            }


            return basePrice + extra;

        }

        // Рендер списка

        // --------------------------------------------------------------

        function renderItems() {

            if (items.length === 0) {

                itemsBody.innerHTML = `

                    <tr>

                        <td colspan="7">

                            <div class="empty-state">

                                <div class="icon">📭</div>

                                <div>Нет добавленных изделий</div>

                                <div style="font-size:13px; margin-top:6px;">Заполните форму выше и нажмите «Добавить»</div>

                            </div>

                        </td>

                    </tr>

                `;

                itemsFoot.style.display = 'none';

                itemsCount.textContent = '0';

                return;

            }


            let html = '';

            let total = 0;


            items.forEach((item, index) => {

                const price = calculatePrice(item.type, item.cat, item.width, item.height);

                total += price;


                const typeName = getTypeName(item.type);


                html += `

                    <tr>

                        <td>${index + 1}</td>

                        <td><span class="badge">${typeName}</span></td>

                        <td>${item.cat}</td>

                        <td>${item.width} см</td>

                        <td>${item.height} см</td>

                        <td><strong>${price.toFixed(2)} $</strong></td>

                        <td>

                            <button class="btn btn-danger btn-sm" onclick="removeItem(${item.id})">✕</button>

                        </td>

                    </tr>

                `;

            });


            // Итоговая строка

            html += `

                <tr class="total-row">

                    <td colspan="5" style="text-align:right;">ИТОГО:</td>

                    <td><strong>${total.toFixed(2)} $</strong></td>

                    <td></td>

                </tr>

            `;


            itemsBody.innerHTML = html;

            itemsFoot.style.display = 'table-row-group';

            itemsCount.textContent = items.length;

            totalPrice.textContent = total.toFixed(2) + ' $';

        }


        function getTypeName(typeKey) {

            const map = {

                'roll_mini': 'Рулонные мини',

                'roll_uni_mgs': 'Рулонные Уни/МГС',

                'dn_mini': 'День-Ночь мини',

                'dn_uni_mgs': 'День-Ночь Уни/МГС'

            };

            return map[typeKey] || typeKey;

        }


        // --------------------------------------------------------------

        // Добавление / удаление

        // --------------------------------------------------------------

        function addItem() {

            const type = typeSelect.value;

            const cat = parseInt(catSelect.value, 10);

            const width = parseInt(widthInput.value, 10) || 80;

            const height = parseInt(heightInput.value, 10) || 170;


            // Валидация

            if (width < 30 || width > 200) {

                alert('Ширина должна быть от 30 до 200 см');

                return;

            }

            if (height < 50 || height > 250) {

                alert('Высота должна быть от 50 до 250 см');

                return;

            }


            items.push({

                id: nextId++,

                type: type,

                cat: cat,

                width: width,

                height: height

            });


            renderItems();

        }


        window.removeItem = function(id) {

            items = items.filter(item => item.id !== id);

            renderItems();

        };


        function clearAll() {

            if (items.length === 0) return;

            if (confirm('Удалить все изделия?')) {

                items = [];

                nextId = 1;

                renderItems();

            }

        }


        // --------------------------------------------------------------

        // События

        // --------------------------------------------------------------

        addBtn.addEventListener('click', addItem);


        // Добавление по Enter

        document.querySelectorAll('#addForm input, #addForm select').forEach(el => {

            el.addEventListener('keydown', (e) => {

                if (e.key === 'Enter') addItem();

            });

        });


        clearAllBtn.addEventListener('click', clearAll);


        // --------------------------------------------------------------

        // Старт

        // --------------------------------------------------------------

        renderItems();

    })();
